function telegram(method,payload){
  try{
    const response=UrlFetchApp.fetch("https://api.telegram.org/bot"+TELEGRAM_BOT_TOKEN+"/"+method,{method:"post",payload:payload,muteHttpExceptions:true}),code=response.getResponseCode(),body=response.getContentText(),result=JSON.parse(body);
    if(!result.ok&&String(result.description||"").indexOf("message is not modified")===-1)Logger.log("Telegram "+method+": "+(result.description||"HTTP "+code));
    return result;
  }catch(e){
    Logger.log("Telegram "+method+": "+(e&&e.message?e.message:String(e)));
    return{ok:false,description:"Ошибка связи с Telegram"};
  }
}

function crmSendMessage(chatId,text,keyboard){
  const payload={chat_id:String(chatId),text:text,parse_mode:"HTML"};
  if(keyboard)payload.reply_markup=JSON.stringify(keyboard);
  const result=telegram("sendMessage",payload);
  return result.ok&&result.result?String(result.result.message_id):"";
}

function deleteTelegramMessage(chatId,messageId){

}

function editTelegramText(chatId,messageId,text,keyboard){

  if(keyboard)payload.reply_markup=JSON.stringify(keyboard);
  const result=telegram("editMessageText",payload);
  return result.ok||String(result.description||"").indexOf("message is not modified")!==-1;
}

}

function sendActiveMessage(chatId,text,keyboard){
  const oldMessageId=getActiveMessage(chatId),canEdit=!keyboard||!!keyboard.inline_keyboard;
  if(oldMessageId&&canEdit&&editTelegramText(chatId,oldMessageId,text,keyboard))return String(oldMessageId);
  const newMessageId=crmSendMessage(chatId,text,keyboard);
  if(!newMessageId)return"";
  setActiveMessage(chatId,newMessageId);
  if(oldMessageId)deleteTelegramMessage(chatId,oldMessageId);
  return String(newMessageId);
}

function sendNewActiveMessage(chatId,text,keyboard,state){
  const messageId=crmSendMessage(chatId,text,keyboard);
  if(!messageId)return"";
  if(state===undefined)setActiveMessage(chatId,messageId);
  else setStateAndActiveMessage(chatId,state,messageId);
  return String(messageId);
}

function replaceActiveMessage(chatId,text,keyboard,state){
  const oldMessageId=getActiveMessage(chatId),newMessageId=crmSendMessage(chatId,text,keyboard);
  if(!newMessageId)return"";
  if(state===undefined)setActiveMessage(chatId,newMessageId);
  else setStateAndActiveMessage(chatId,state,newMessageId);
  if(oldMessageId)deleteTelegramMessage(chatId,oldMessageId);
  return String(newMessageId);
}
