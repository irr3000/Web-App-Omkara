function isTelegramUrl(value){
  return/^(https?:\/\/|tg:\/\/)/i.test(String(value||"").trim());
}

function normalizeEmail(value){
  return String(value||"").trim().toLowerCase();
}

function isValidEmail(value){
  return value.length<=254&&/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(value);
}

function escapeHtml(value){
  return String(value==null?"":value).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/"/g,"&quot;");
}

function getConfigurationErrors(){
  const errors=Database.getAdapterErrors();
  if(!errors.length){
    const schemaErrors=Database.validateSchema(["users","applications","retreats"]);
    for(let i=0;i<schemaErrors.length;i++)errors.push(schemaErrors[i]);
  }
  if(!TELEGRAM_BOT_TOKEN)errors.push("Не задано свойство TELEGRAM_BOT_TOKEN");
  if(!VK_ACCESS_TOKEN)errors.push("Не задано свойство VK_ACCESS_TOKEN");
  if(!WEBHOOK_SECRET)errors.push("Не задано свойство WEBHOOK_SECRET");
  return errors;
}

function validateConfiguration(){
  const errors=getConfigurationErrors(),text=errors.length?"⚠️ <b>Проверка CRM</b>\n\n"+errors.map(function(item){return"• "+escapeHtml(item);}).join("\n"):"✅ <b>Проверка CRM</b>\n\nКонфигурация заполнена правильно.";
  Logger.log(text.replace(/<[^>]+>/g,""));
  if(TELEGRAM_BOT_TOKEN)crmSendMessage(ADMIN_CHAT_ID,text);
  return errors.length===0;
}
