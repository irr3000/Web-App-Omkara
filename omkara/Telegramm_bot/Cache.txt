/* Универсальная инфраструктура кэша. Не зависит от конкретного хранилища данных. */
var AppCache=(function(){
  var UPDATE_PREFIX="telegram_update_",UPDATE_SECONDS=600;
  var runtimes=Object.create(null);

  function scriptCache(){
    return CacheService.getScriptCache();
  }

  function read(key){
    return scriptCache().get(String(key));
  }

  function write(key,value,seconds){
    scriptCache().put(String(key),String(value),seconds);
  }

  function remove(key){
    scriptCache().remove(String(key));
  }

  function createPersistentStore(prefix,seconds,emptySeconds){
    prefix=String(prefix||"");
    return{
      get:function(key){return read(prefix+String(key));},
      set:function(key,value){write(prefix+String(key),value,value?seconds:(emptySeconds||seconds));},
      remove:function(key){remove(prefix+String(key));}
    };
  }

  function getRuntime(namespace,factory){
    namespace=String(namespace||"default");
    if(!Object.prototype.hasOwnProperty.call(runtimes,namespace))runtimes[namespace]=factory?factory():Object.create(null);
    return runtimes[namespace];
  }

  function resetRuntime(){
    var keys=Object.keys(runtimes);
    for(var i=0;i<keys.length;i++)delete runtimes[keys[i]];
  }

  function wasUpdateProcessed(updateId){
    return!!read(UPDATE_PREFIX+String(updateId));
  }

  function markUpdateProcessed(updateId){
    write(UPDATE_PREFIX+String(updateId),"1",UPDATE_SECONDS);
  }

  function forgetUpdate(updateId){
    remove(UPDATE_PREFIX+String(updateId));
  }

  return{
    createPersistentStore:createPersistentStore,
    getRuntime:getRuntime,
    resetRuntime:resetRuntime,
    wasUpdateProcessed:wasUpdateProcessed,
    markUpdateProcessed:markUpdateProcessed,
    forgetUpdate:forgetUpdate
  };
})();
